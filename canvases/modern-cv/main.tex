% --- Background for pages (drawn at shipout) ---
\AddToShipoutPictureBG{%
  \ifwithsidebar
    \begin{tikzpicture}[remember picture,overlay]
      % Sidebar background full height
      \fill[SidebarBG] (current page.north west)
        rectangle ([xshift=\SidebarW]current page.south west);

      % Header + photo + titres (seulement si header actif)
      \ifwithheader
        \ifnum\value{page}=1
        % Header bar (plus chevauchant la sidebar)
        \coordinate (barleft) at ([xshift=\dimexpr\SidebarW-\CVHeaderBarLeftOverlap\relax,yshift=\CVHeaderBarTopShift]current page.north west);
        \coordinate (barright) at ([yshift=\CVHeaderBarBottomShift]current page.north east);
        \fill[HeaderBG] (barleft) rectangle (barright);

        % Centre horizontal de la zone blanche (sans sidebar)
        \coordinate (whitecenter) at ($([xshift=\SidebarW]current page.north west)!0.5!(current page.north east)$);
        \coordinate (barverticalcenter) at ($(barleft)!0.5!(barright)$);
        \coordinate (barcenter) at (whitecenter |- barverticalcenter);

        % Photo (plus grande, à cheval)
        \coordinate (photoc) at ([xshift=\dimexpr\Pad+\CVPhotoXFromPad\relax,yshift=\CVPhotoYShift]current page.north west);
        \def\photor{\CVPhotoRadius}

        \begin{scope}
          \clip (photoc) circle (\photor);
          \node at (photoc) {\includegraphics[width=\CVPhotoWidth]{\CVPhotoPath}};
        \end{scope}
        \draw[white,line width=\CVPhotoBorderWidth] (photoc) circle (\photor);

        % Name + title
\node[anchor=center, align=center] at (barcenter) {%
  {\color{white}\bfseries\fontsize{\CVHeaderNameSize}{\CVHeaderNameSize}\selectfont
   \addfontfeatures{LetterSpace=\CVHeaderNameLetterSpace} \CVHeaderName}\\[\CVHeaderNameTitleGap]
  {\color{white!75}\large \CVHeaderTitle}%
};
        \fi
      \fi
    \end{tikzpicture}
  \fi
}
